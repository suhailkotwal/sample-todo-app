# app/services/ai_agent.py
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.prompts import ChatPromptTemplate
from langchain.agents import create_tool_calling_agent, AgentExecutor
from fuzzywuzzy import fuzz
from sqlalchemy.orm import Session
from ..models.client import Client, FinancialData, Task
import os
from typing import List, Optional
from datetime import datetime

class ClientInfoAgent:
    def __init__(self, db_session: Session):
        self.db = db_session
        self.llm = ChatOpenAI(
            model="gpt-3.5-turbo",
            temperature=0,
            openai_api_key=os.getenv("OPENAI_API_KEY")
        )
        
        # Create tools for the agent
        self.tools = [
            self._create_search_client_tool(),
            self._create_get_financial_data_tool(),
            self._create_assess_risk_tool(),
            self._create_find_missing_data_tool(),
            self._create_check_compliance_tool()
        ]
        
        # Create the agent
        self.agent_executor = self._create_agent()
    
    def _create_search_client_tool(self):
        @tool
        def search_client(client_name: str) -> str:
            """Search for a client by name using fuzzy matching. Use this when user mentions a client name."""
            clients = self.db.query(Client).all()
            best_match = None
            best_score = 0
            
            for client in clients:
                # Check main name
                score = fuzz.ratio(client_name.lower(), client.name.lower())
                
                # Check aliases
                if client.aliases:
                    for alias in client.aliases:
                        alias_score = fuzz.ratio(client_name.lower(), alias.lower())
                        score = max(score, alias_score)
                
                if score > best_score and score > 70:
                    best_score = score
                    best_match = client
            
            if best_match:
                return f"Found client: {best_match.name} (ID: {best_match.id}, Industry: {best_match.industry}, Risk Score: {best_match.risk_score})"
            else:
                return f"No client found matching '{client_name}'. Available clients: {', '.join([c.name for c in clients[:5]])}"
        
        return search_client
    
    def _create_get_financial_data_tool(self):
        @tool
        def get_financial_data(client_name: str) -> str:
            """Get financial data for a specific client. Use after finding the client."""
            client = self._find_client_by_name(client_name)
            if not client:
                return f"Client '{client_name}' not found"
            
            financial_data = self.db.query(FinancialData).filter(
                FinancialData.client_id == client.id
            ).order_by(FinancialData.period).all()
            
            if not financial_data:
                return f"No financial data found for {client.name}"
            
            result = f"Financial data for {client.name}:\n"
            for data in financial_data:
                result += f"- {data.period}: Revenue ${data.revenue:,.0f}, Profit ${data.profit:,.0f}, Debt Ratio {data.debt_ratio:.1%}\n"
            
            return result
        
        return get_financial_data
    
    def _create_assess_risk_tool(self):
        @tool
        def assess_credit_risk(client_name: str) -> str:
            """Assess credit risk for a client based on their financial data and completeness."""
            client = self._find_client_by_name(client_name)
            if not client:
                return f"Client '{client_name}' not found"
            
            financial_data = self.db.query(FinancialData).filter(
                FinancialData.client_id == client.id
            ).order_by(FinancialData.period.desc()).all()
            
            if not financial_data:
                return f"Cannot assess risk for {client.name} - no financial data available"
            
            # Risk assessment logic
            risk_factors = []
            latest_data = financial_data[0]
            
            # Check debt ratio
            if latest_data.debt_ratio and latest_data.debt_ratio > 0.5:
                risk_factors.append("High debt ratio (>50%)")
            
            # Check profitability
            if latest_data.profit and latest_data.profit < 0:
                risk_factors.append("Recent losses")
            
            # Check data completeness for 2024
            current_year_periods = [d.period for d in financial_data if "2024" in d.period]
            expected_2024_periods = ["2024-Q1", "2024-Q2", "2024-Q3", "2024-Q4"]
            missing_2024 = [p for p in expected_2024_periods if p not in current_year_periods]
            
            if len(missing_2024) > 2:
                risk_factors.append("Incomplete 2024 financial data")
            
            # Determine risk level
            if len(risk_factors) == 0:
                risk_level = "Low"
            elif len(risk_factors) <= 2:
                risk_level = "Moderate" 
            else:
                risk_level = "High"
            
            result = f"Credit risk assessment for {client.name}: {risk_level} risk\n"
            result += f"Current risk score: {client.risk_score:.1%}\n"
            
            if risk_factors:
                result += f"Risk factors: {', '.join(risk_factors)}\n"
            else:
                result += "No significant risk factors identified\n"
            
            return result
        
        return assess_credit_risk
    
    def _create_find_missing_data_tool(self):
        @tool
        def find_missing_data(client_name: str) -> str:
            """Find missing financial periods for a client and create tasks if needed."""
            client = self._find_client_by_name(client_name)
            if not client:
                return f"Client '{client_name}' not found"
            
            # Get existing financial data
            financial_data = self.db.query(FinancialData).filter(
                FinancialData.client_id == client.id
            ).all()
            
            existing_periods = [fd.period for fd in financial_data]
            
            # Expected periods
            expected_periods = ["2023", "2024-Q1", "2024-Q2", "2024-Q3", "2024-Q4"]
            missing_periods = [p for p in expected_periods if p not in existing_periods]
            
            if not missing_periods:
                return f"All expected financial data is available for {client.name}"
            
            # Create tasks for missing periods (if they don't already exist)
            created_tasks = []
            for period in missing_periods:
                existing_task = self.db.query(Task).filter(
                    Task.client_id == client.id,
                    Task.task_type == "financial_data",
                    Task.description.contains(period),
                    Task.status == "pending"
                ).first()
                
                if not existing_task:
                    priority = "high" if "2024" in period else "medium"
                    new_task = Task(
                        client_id=client.id,
                        description=f"Obtain {period} financial data for {client.name}",
                        priority=priority,
                        task_type="financial_data"
                    )
                    self.db.add(new_task)
                    created_tasks.append(period)
            
            if created_tasks:
                self.db.commit()
            
            result = f"Missing financial data for {client.name}: {', '.join(missing_periods)}\n"
            if created_tasks:
                result += f"Created tasks for: {', '.join(created_tasks)}\n"
            result += f"Existing periods: {', '.join(existing_periods)}"
            
            return result
        
        return find_missing_data
    
    def _create_check_compliance_tool(self):
        @tool
        def check_compliance_flags(client_name: str) -> str:
            """Check compliance issues and flags for a client."""
            client = self._find_client_by_name(client_name)
            if not client:
                return f"Client '{client_name}' not found"
            
            compliance_issues = []
            
            # Check financial data currency
            financial_data = self.db.query(FinancialData).filter(
                FinancialData.client_id == client.id
            ).all()
            
            if not financial_data:
                compliance_issues.append("No financial records on file")
            else:
                # Check for current year data
                current_year_data = [fd for fd in financial_data if "2024" in fd.period]
                if not current_year_data:
                    compliance_issues.append("Missing current year (2024) financial data")
            
            # Check risk score thresholds
            if client.risk_score > 0.7:
                compliance_issues.append("High risk score (>70%) requires enhanced monitoring")
            
            # Check for pending compliance tasks
            compliance_tasks = self.db.query(Task).filter(
                Task.client_id == client.id,
                Task.task_type.contains("compliance"),
                Task.status == "pending"
            ).all()
            
            if compliance_tasks:
                compliance_issues.append(f"{len(compliance_tasks)} pending compliance task(s)")
            
            if not compliance_issues:
                return f"No compliance flags found for {client.name}. All regulatory requirements appear up to date."
            else:
                result = f"Compliance flags for {client.name}:\n"
                for i, issue in enumerate(compliance_issues, 1):
                    result += f"{i}. {issue}\n"
                return result
        
        return check_compliance_flags
    
    def _create_agent(self):
        """Create the LangChain agent with tools"""
        prompt = ChatPromptTemplate.from_messages([
            ("system", """You are a credit officer assistant helping with client information and risk assessment.

Available tools:
- search_client: Find clients by name (use fuzzy matching)
- get_financial_data: Get financial information for a client
- assess_credit_risk: Calculate risk assessment based on data
- find_missing_data: Identify gaps in financial data and create tasks
- check_compliance_flags: Check regulatory compliance status

Instructions:
1. Always search for the client first when a name is mentioned
2. Be specific and provide actionable information
3. When data is missing, mention what tasks have been created
4. Format financial numbers clearly (e.g., $1,000,000)
5. Explain risk factors clearly
6. If you can't find a client, suggest similar names from available clients

Always provide helpful, professional responses that a credit officer would find valuable."""),
            ("placeholder", "{chat_history}"),
            ("human", "{input}"),
            ("placeholder", "{agent_scratchpad}")
        ])
        
        agent = create_tool_calling_agent(self.llm, self.tools, prompt)
        return AgentExecutor(agent=agent, tools=self.tools, verbose=False)
    
    def process_query(self, user_query: str) -> str:
        """Process a natural language query and return response"""
        try:
            result = self.agent_executor.invoke({"input": user_query})
            return result["output"]
        except Exception as e:
            return f"I apologize, but I encountered an error processing your query: {str(e)}"
    
    def _find_client_by_name(self, client_name: str) -> Optional[Client]:
        """Helper function to find client by fuzzy name matching"""
        clients = self.db.query(Client).all()
        
        for client in clients:
            if fuzz.ratio(client_name.lower(), client.name.lower()) > 70:
                return client
            
            if client.aliases:
                for alias in client.aliases:
                    if fuzz.ratio(client_name.lower(), alias.lower()) > 70:
                        return client
        
        return None

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from pydantic import BaseModel
from ..models.database import get_db
from ..services.ai_agent import ClientInfoAgent
import time

router = APIRouter()

class QueryRequest(BaseModel):
    query: str

class QueryResponse(BaseModel):
    response: str
    query: str
    processing_time: float

@router.post("/query")
async def process_query(request: QueryRequest, db: Session = Depends(get_db)):
    """Process natural language queries about clients"""
    try:
        start_time = time.time()
        
        # Initialize AI agent
        agent = ClientInfoAgent(db)
        
        # Process the query
        response = agent.process_query(request.query)
        
        processing_time = time.time() - start_time
        
        return QueryResponse(
            response=response,
            query=request.query,
            processing_time=round(processing_time, 2)
        )
        
    except Exception as e:
        raise HTTPException(
            status_code=500, 
            detail=f"Error processing query: {str(e)}"
        )

    """Create a new client"""

@app.get("/clients/{client_id}")
async def get_client(client_id: int, db: Session = Depends(get_db)):
    """Get specific client with financial data and tasks"""
    client = db.query(Client).filter(Client.id == client_id).first()
    
    if not client:
        raise HTTPException(status_code=404, detail="Client not found")
    
    return {
        "id": client.id,
        "name": client.name,
        "industry": client.industry,
        "risk_score": client.risk_score,
        "aliases": client.aliases or [],
        "financial_data": [
            {
                "period": fd.period,
                "revenue": fd.revenue,
                "profit": fd.profit,
                "debt_ratio": fd.debt_ratio
            }
            for fd in client.financial_data
        ],
        "tasks": [
            {
                "id": task.id,
                "description": task.description,
                "priority": task.priority,
                "status": task.status,
                "created_at": task.created_at.isoformat()
            }
            for task in client.tasks
        ]
    }

@app.get("/tasks")
async def get_tasks(status: str = None, db: Session = Depends(get_db)):
    """Get all tasks, optionally filter by status"""
    query = db.query(Task).join(Client)
    
    if status:
        query = query.filter(Task.status == status)
    
    tasks = query.all()
    
    return [
        {
            "id": task.id,
            "client_name": task.client.name,
            "description": task.description,
            "priority": task.priority,
            "status": task.status,
            "created_at": task.created_at.isoformat()
        }
        for task in tasks
    ]
