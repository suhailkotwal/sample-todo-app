# app/main.py - Updated with AI capabilities
from fastapi import FastAPI, Depends, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from sqlalchemy.orm import Session
from dotenv import load_dotenv
from typing import List
import os

# Import database components
from .models.database import get_db, create_tables
from .models.client import Client, FinancialData, Task

# Import API routers
from .api.queries import router as queries_router

load_dotenv()

app = FastAPI(
    title="Client Info Assistant",
    description="AI-powered client data management system",
    version="0.3.0"
)

# Enable CORS
app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:3000", "http://localhost:5173"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# Include API routers
app.include_router(queries_router, prefix="/api", tags=["AI Queries"])

# Startup event
@app.on_event("startup")
async def startup_event():
    print("🚀 Starting Client Info Assistant API...")
    create_tables()
    
    # Check if Azure OpenAI is configured
    azure_configured = all([
        os.getenv("AZURE_OPENAI_API_KEY"),
        os.getenv("AZURE_OPENAI_ENDPOINT"), 
        os.getenv("AZURE_OPENAI_DEPLOYMENT_NAME"),
        os.getenv("AZURE_OPENAI_API_VERSION")
    ]) and not any([
        os.getenv("AZURE_OPENAI_API_KEY") == "your_azure_openai_api_key_here",
        os.getenv("AZURE_OPENAI_ENDPOINT") == "https://your-resource-name.openai.azure.com/"
    ])
    
    if not azure_configured:
        print("⚠️  WARNING: Azure OpenAI not configured! Update your .env file with:")
        print("   AZURE_OPENAI_API_KEY=your_key")
        print("   AZURE_OPENAI_ENDPOINT=https://your-resource.openai.azure.com/")
        print("   AZURE_OPENAI_DEPLOYMENT_NAME=your-deployment")
        print("   AZURE_OPENAI_API_VERSION=2024-02-15-preview")
    else:
        print("✅ Azure OpenAI configured")
    
    print("✅ Database initialized!")
    print("🤖 AI agent ready!")

@app.get("/health")
async def health(db: Session = Depends(get_db)):
    try:
        # Test database connection
        client_count = db.query(Client).count()
        task_count = db.query(Task).count()
        
        # Test Azure OpenAI connection
        azure_status = "not_configured"
        azure_error = None
        
        # Check if Azure OpenAI is configured
        azure_configured = all([
            os.getenv("AZURE_OPENAI_API_KEY"),
            os.getenv("AZURE_OPENAI_ENDPOINT"), 
            os.getenv("AZURE_OPENAI_DEPLOYMENT_NAME"),
            os.getenv("AZURE_OPENAI_API_VERSION")
        ]) and not any([
            os.getenv("AZURE_OPENAI_API_KEY") == "your_azure_openai_api_key_here",
            os.getenv("AZURE_OPENAI_ENDPOINT") == "https://your-resource-name.openai.azure.com/"
        ])
        
        if azure_configured:
            # Test actual connection to Azure OpenAI
            try:
                from openai import AzureOpenAI
                
                client = AzureOpenAI(
                    api_key=os.getenv("AZURE_OPENAI_API_KEY"),
                    api_version=os.getenv("AZURE_OPENAI_API_VERSION"),
                    azure_endpoint=os.getenv("AZURE_OPENAI_ENDPOINT")
                )
                
                # Simple test call
                response = client.chat.completions.create(
                    model=os.getenv("AZURE_OPENAI_DEPLOYMENT_NAME"),
                    messages=[{"role": "user", "content": "Hello"}],
                    max_tokens=5
                )
                
                azure_status = "connected"
                
            except Exception as e:
                azure_status = "connection_failed"
                azure_error = str(e)
        
        health_response = {
            "status": "healthy",
            "database": "connected",
            "clients": client_count,
            "tasks": task_count,
            "azure_openai": {
                "status": azure_status,
                "configured": azure_configured
            }
        }
        
        # Add error details if connection failed
        if azure_error:
            health_response["azure_openai"]["error"] = azure_error
            health_response["azure_openai"]["troubleshooting"] = {
                "check_endpoint": os.getenv("AZURE_OPENAI_ENDPOINT"),
                "check_deployment": os.getenv("AZURE_OPENAI_DEPLOYMENT_NAME"),
                "check_api_version": os.getenv("AZURE_OPENAI_API_VERSION")
            }
        
        return health_response
        
    except Exception as e:
        return {
            "status": "error",
            "database": "failed",
            "error": str(e)
        }

# Keep existing CRUD endpoints
@app.get("/clients")
async def get_clients(db: Session = Depends(get_db)):
    """Get all clients"""
    clients = db.query(Client).all()
    return [
        {
            "id": client.id,
            "name": client.name,
            "industry": client.industry,
            "risk_score": client.risk_score,
            "aliases": client.aliases or []
        }
        for client in clients
    ]

@app.post("/clients")
async def create_client(
    name: str,
    industry: str = None,
    aliases: List[str] = None,
    db: Session = Depends(get_db)
):
    """Create a new client"""
    client = Client(
        name=name,
        industry=industry,
        aliases=aliases or [],
        risk_score=0.0
    )
    
    db.add(client)
    db.commit()
    db.refresh(client)
    
    return {
        "id": client.id,
        "name": client.name,
        "industry": client.industry,
        "message": "Client created successfully"
    }

@app.get("/clients/{client_id}")
async def get_client(client_id: int, db: Session = Depends(get_db)):
    """Get specific client with financial data and tasks"""
    client = db.query(Client).filter(Client.id == client_id).first()
    
    if not client:
        raise HTTPException(status_code=404, detail="Client not found")
    
    return {
        "id": client.id,
        "name": client.name,
        "industry": client.industry,
        "risk_score": client.risk_score,
        "aliases": client.aliases or [],
        "financial_data": [
            {
                "period": fd.period,
                "revenue": fd.revenue,
                "profit": fd.profit,
                "debt_ratio": fd.debt_ratio
            }
            for fd in client.financial_data
        ],
        "tasks": [
            {
                "id": task.id,
                "description": task.description,
                "priority": task.priority,
                "status": task.status,
                "created_at": task.created_at.isoformat()
            }
            for task in client.tasks
        ]
    }

@app.get("/tasks")
async def get_tasks(status: str = None, db: Session = Depends(get_db)):
    """Get all tasks, optionally filter by status"""
    query = db.query(Task).join(Client)
    
    if status:
        query = query.filter(Task.status == status)
    
    tasks = query.all()
    
    return [
        {
            "id": task.id,
            "client_name": task.client.name,
            "description": task.description,
            "priority": task.priority,
            "status": task.status,
            "created_at": task.created_at.isoformat()
        }
        for task in tasks
    ]

@app.post("/test/create-sample-data")
async def create_sample_data(db: Session = Depends(get_db)):
    """Create sample data for testing AI queries"""
    
    # Clear existing data
    db.query(Task).delete()
    db.query(FinancialData).delete()
    db.query(Client).delete()
    
    # Create sample clients with more detailed data
    clients_data = [
        {
            "name": "Tech Corp Inc",
            "industry": "Technology",
            "aliases": ["TechCorp", "Tech Corp", "TC Inc"],
            "risk_score": 0.35,
            "financial_data": [
                {"period": "2023", "revenue": 1000000, "profit": 150000, "debt_ratio": 0.3},
                {"period": "2024-Q1", "revenue": 300000, "profit": 45000, "debt_ratio": 0.32}
                # Missing Q2, Q3, Q4 2024
            ]
        },
        {
            "name": "Manufacturing Ltd", 
            "industry": "Manufacturing",
            "aliases": ["Mfg Ltd", "Manufacturing Limited"],
            "risk_score": 0.65,
            "financial_data": [
                {"period": "2023", "revenue": 2500000, "profit": 200000, "debt_ratio": 0.45}
                # Missing all 2024 data
            ]
        },
        {
            "name": "Retail Solutions",
            "industry": "Retail", 
            "aliases": ["RetailSol", "RS Inc"],
            "risk_score": 0.15,
            "financial_data": [
                {"period": "2023", "revenue": 800000, "profit": 120000, "debt_ratio": 0.25},
                {"period": "2024-Q1", "revenue": 220000, "profit": 35000, "debt_ratio": 0.23},
                {"period": "2024-Q2", "revenue": 250000, "profit": 40000, "debt_ratio": 0.22}
                # Missing Q3, Q4 2024
            ]
        },
        {
            "name": "StartupXYZ",
            "industry": "Technology",
            "aliases": ["Startup XYZ", "XYZ Co"],
            "risk_score": 0.85,  # High risk
            "financial_data": [
                {"period": "2024-Q1", "revenue": 50000, "profit": -30000, "debt_ratio": 0.8}  # Losses
            ]
        }
    ]
    
    created_clients = []
    
    for client_data in clients_data:
        # Create client
        financial_data = client_data.pop("financial_data")
        client = Client(**client_data)
        db.add(client)
        db.commit()
        db.refresh(client)
        created_clients.append(client)
        
        # Add financial data
        for fd_data in financial_data:
            fd = FinancialData(client_id=client.id, **fd_data)
            db.add(fd)
        
        # Create tasks for missing data
        existing_periods = [fd["period"] for fd in financial_data]
        expected_periods = ["2023", "2024-Q1", "2024-Q2", "2024-Q3", "2024-Q4"]
        missing_periods = [p for p in expected_periods if p not in existing_periods]
        
        for period in missing_periods:
            priority = "high" if "2024" in period else "medium"
            task = Task(
                client_id=client.id,
                description=f"Obtain {period} financial data for {client.name}",
                priority=priority,
                task_type="financial_data"
            )
            db.add(task)
    
    db.commit()
    
    return {
        "message": "Enhanced sample data created successfully",
        "clients_created": len(created_clients),
        "client_names": [c.name for c in created_clients],
        "ai_ready": True,
        "try_queries": [
            "What is the credit risk for Tech Corp?",
            "Show financial data for Manufacturing Ltd",
            "What's missing for StartupXYZ?"
        ]
    }
