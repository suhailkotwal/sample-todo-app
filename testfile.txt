def process_query_with_history(self, user_query: str, chat_history: List = None) -> str:
        """Process a natural language query with conversation history"""
        try:
            # Format chat history for LangChain agent
            formatted_history = []
            if chat_history:
                # Take last 10 messages to avoid token limits
                recent_history = chat_history[-10:] if len(chat_history) > 10 else chat_history
                
                for msg in recent_history:
                    if hasattr(msg, 'role') and hasattr(msg, 'content'):
                        # Handle Pydantic models
                        role = msg.role
                        content = msg.content
                    elif isinstance(msg, dict):
                        # Handle dictionaries
                        role = msg.get('role')
                        content = msg.get('content')
                    elif isinstance(msg, tuple) and len(msg) == 2:
                        # Handle tuples (role, content)
                        role, content = msg
                    else:
                        continue
                    
                    # Convert to LangChain format
                    if role == "user":
                        formatted_history.append(("human", content))
                    elif role == "assistant":
                        formatted_history.append(("ai", content))
            
            # Process with history
            result = self.agent_executor.invoke({
                "input": user_query,
                "chat_history": formatted_history
            })
            
            return result["output"]
        except Exception as e:
            return f"I apologize, but I encountered an error processing your query: {str(e)}"
