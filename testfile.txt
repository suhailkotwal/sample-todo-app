# app/main.py
from fastapi import FastAPI, HTTPException, Depends
from fastapi.middleware.cors import CORSMiddleware
from sqlalchemy.orm import Session
import os
from dotenv import load_dotenv

from .models.database import get_db, create_tables
from .api import clients, queries, tasks
from .data.startup_loader import populate_database_on_startup

load_dotenv()

# Create FastAPI app
app = FastAPI(
    title="Client Info Assistant API",
    description="AI-powered client information and risk assessment system",
    version="1.0.0"
)

# CORS middleware for frontend communication
allowed_origins = os.getenv("ALLOWED_ORIGINS", "http://localhost:3000").split(",")
app.add_middleware(
    CORSMiddleware,
    allow_origins=allowed_origins,
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# Startup event
@app.on_event("startup")
async def startup_event():
    """Initialize database and load demo data on startup"""
    print("Starting Client Info Assistant API...")
    create_tables()
    populate_database_on_startup()
    print("API ready!")

# Health check endpoint
@app.get("/")
async def root():
    return {
        "message": "Client Info Assistant API",
        "status": "running",
        "version": "1.0.0"
    }

@app.get("/health")
async def health_check():
    return {"status": "healthy"}

# Include routers
app.include_router(clients.router, prefix="/api", tags=["clients"])
app.include_router(queries.router, prefix="/api", tags=["queries"])
app.include_router(tasks.router, prefix="/api", tags=["tasks"])

# -------------------------------------------------------------------
# run.py (in backend root)
import uvicorn
import os
from dotenv import load_dotenv

load_dotenv()

if __name__ == "__main__":
    host = os.getenv("APP_HOST", "0.0.0.0")
    port = int(os.getenv("APP_PORT", 8000))
    debug = os.getenv("DEBUG", "True").lower() == "true"
    
    uvicorn.run(
        "app.main:app",
        host=host,
        port=port,
        reload=debug
    )
